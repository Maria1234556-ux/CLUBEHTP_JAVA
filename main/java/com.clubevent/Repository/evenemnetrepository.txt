package com.clubevents.repository;

import com.clubevents.model.Evenement;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import java.util.List;

public interface EvenementRepository extends JpaRepository<Evenement, Long> {
    List<Evenement> findByOuvertATousTrue();

    @Query("SELECT e FROM Evenement e WHERE e.club.id IN :clubIds")
    List<Evenement> findByClubIds(@Param("clubIds") List<Long> clubIds);
}
