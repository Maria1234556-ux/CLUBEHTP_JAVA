package com.clubevents.model;

import javax.persistence.*;
import java.util.ArrayList;
import java.util.List;

@Entity
public class Utilisateur extends Personne {
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    private int age;
    private String filiere;
    private String annee;
    private String role;

    @OneToMany(mappedBy = "utilisateur", cascade = CascadeType.ALL)
    private List<MembreClub> clubs = new ArrayList<>();

    public Utilisateur() {
        super();
    }

    public Utilisateur(String nom, String prenom, int age, String filiere, String annee) {
        super(nom, prenom);
        this.age = age;
        this.filiere = filiere;
        this.annee = annee;
    }

    // Polymorphisme - Redéfinition de méthode
    @Override
    public String getDescription() {
        return "Utilisateur: " + nom + " " + prenom + " (" + role + ")";
    }

    @Override
    public String toString() {
        return nom + " " + prenom;
    }

    // Getters
    public Long getId() { return id; }
    public int getAge() { return age; }
    public String getFiliere() { return filiere; }
    public String getAnnee() { return annee; }
    public String getRole() { return role; }
    public List<MembreClub> getClubs() { return clubs; }

    // Setters
    public void setId(Long id) { this.id = id; }
    public void setAge(int age) { this.age = age; }
    public void setFiliere(String filiere) { this.filiere = filiere; }
    public void setAnnee(String annee) { this.annee = annee; }
    public void setRole(String role) { this.role = role; }
    public void setClubs(List<MembreClub> clubs) { this.clubs = clubs; }
}
