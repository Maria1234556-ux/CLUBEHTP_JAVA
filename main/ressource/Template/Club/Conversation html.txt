<script>
    const urlParams = new URLSearchParams(window.location.search);
    const celluleId = urlParams.get('celluleId');
    const cellName = urlParams.get('cellName') || "Cell";
    const userRole = urlParams.get('role') || 'MEMBRE';

    // Set cell name and info
    document.getElementById('cellName').innerText = cellName;

    const cellsInfo = {
        1: "Media & Design Cell - 14 members | Leader: Ahmed",
        2: "Logistic Cell - 14 members | Leader: Fatima",
        3: "Event Cell - 14 members | Leader: Karim",
        4: "Sponsoring Cell - 14 members | Leader: Fouad",
        5: "Project Cell - 14 members | Leader: Mouad",
        6: "Course Cell - 14 members | Leader: Laila"
    };

    document.getElementById('cellInfo').innerText = cellsInfo[celluleId] || "14 members";

    function sendMessage() {
        const input = document.getElementById('messageInput');
        const message = input.value.trim();

        if (message) {
            const container = document.getElementById('messagesContainer');
            const now = new Date();
            const time = now.getHours() + ':' + String(now.getMinutes()).padStart(2, '0');

            const newMessage = document.createElement('div');
            newMessage.className = 'message message-sent';
            newMessage.innerHTML = `
                <div class="message-bubble">
                    ${message.replace(/</g, '&lt;')}
                </div>
                <div class="message-info">You • ${time}</div>
            `;
            container.appendChild(newMessage);
            input.value = '';
            container.scrollTop = container.scrollHeight;
        }
    }

    document.getElementById('messageInput').addEventListener('keypress', function(e) {
        if (e.key === 'Enter') {
            sendMessage();
        }
    });

    const container = document.getElementById('messagesContainer');
    container.scrollTop = container.scrollHeight;
</script>
