<script>
    // Récupérer l'ID du club depuis l'URL
    const urlParams = new URLSearchParams(window.location.search);
    const clubId = urlParams.get('clubId');
    const clubName = urlParams.get('clubName');

    if (clubName) {
        document.getElementById('clubName').innerText = clubName;
    }

    function openModal() {
        document.getElementById('createModal').style.display = 'block';
    }

    function closeModal() {
        document.getElementById('createModal').style.display = 'none';
    }

    function createCellule() {
        const name = document.getElementById('celluleName').value;
        const description = document.getElementById('celluleDesc').value;

        if (name && description) {
            // Ajouter la nouvelle cellule à la grille
            const container = document.getElementById('cellulesContainer');
            const newId = container.children.length + 1;

            const newCellule = document.createElement('div');
            newCellule.className = 'cellule-card';
            newCellule.onclick = () => showConversation(newId);
            newCellule.innerHTML = `
                <div class="cellule-header">
                    <h3>${name}</h3>
                    <span class="chef-badge">Chef: Vous</span>
                </div>
                <div class="cellule-description">
                    ${description}
                </div>
                <div class="cellule-stats">
                    <span>👥 1 membres</span>
                    <span>💬 0 messages</span>
                </div>
            `;
            container.appendChild(newCellule);

            closeModal();
            document.getElementById('celluleName').value = '';
            document.getElementById('celluleDesc').value = '';
        }
    }

    function showConversation(celluleId) {
        window.location.href = `/clubs/conversation?celluleId=${celluleId}&clubId=${clubId}&clubName=${clubName}`;
    }

    // Fermer le modal en cliquant en dehors
    window.onclick = function(event) {
        const modal = document.getElementById('createModal');
        if (event.target == modal) {
            closeModal();
        }
    }
    // Récupérer le rôle de l'utilisateur (depuis une API ou session)
let userRole = 'MEMBRE'; // À remplacer par la valeur réelle

// Vérifier si l'utilisateur est du bureau
function isBureau() {
    return userRole === 'BUREAU';
}

// Afficher ou masquer le bouton de création
function updateButtons() {
    const createBtn = document.querySelector('.btn-create');
    if (createBtn) {
        if (isBureau()) {
            createBtn.style.display = 'block';
        } else {
            createBtn.style.display = 'none';
        }
    }
}
</script>
