    document.getElementById('userForm').addEventListener('submit', async function(e) {
        e.preventDefault();

        const nom = document.getElementById('nom').value;
        const prenom = document.getElementById('prenom').value;
        const age = document.getElementById('age').value;
        const filiere = document.getElementById('filiere').value;
        const annee = document.getElementById('annee').value;

        const userResponse = await fetch('/api/inscription', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
        nom: nom,
        prenom: prenom,
        age: parseInt(age),
        filiere: filiere,
        annee: annee,        // ← IL MANQUE LA VIRGULE ICI !
        role: document.getElementById('role').value
    })
});

        const userId = await userResponse.json();

        const clubSelects = document.querySelectorAll('.club-nom');
        const positionSelects = document.querySelectorAll('.club-position');

        for(let i = 0; i < clubSelects.length; i++) {
            const clubNom = clubSelects[i].value;
            const position = positionSelects[i].value;

            if(clubNom && position) {
                await fetch('/api/ajouter-club?userId=' + userId + '&clubNom=' + encodeURIComponent(clubNom) + '&position=' + encodeURIComponent(position), {
                    method: 'POST'
                });
            }
        }

        window.location.href = '/dashboard/' + userId;
    });
